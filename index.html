function showToast(msg) {
const toast = document.getElementById('toast');
document.getElementById('toastMsg').textContent = msg;
toast.classList.remove('hidden');
clearTimeout(window.toastTimer);
window.toastTimer = setTimeout(() => toast.classList.add('hidden'), 2400);
}

function syncSearch(val) {

renderProducts();
}

function renderCategories(products) {
const cats = ['All', ...new Set(products.map(p => p.category).filter(Boolean))];
const container = document.getElementById('categoryTabs');
container.innerHTML = cats.map(cat => {
const active = selectedCategory === cat;
return `<button onclick="selectCategory('${cat.replace(/'/g, "\\'")}')"
class="px-3.5 py-1.5 rounded-full text-xs font-bold whitespace-nowrap transition ${
active ? 'bg-slate-900 text-white shadow-sm' : 'bg-white text-slate-600 border border-slate-200 hover:border-slate-300'
}">${cat === 'All' ? 'সবগুলো' : cat}</button>`;
}).join('');
}

function selectCategory(cat) {
selectedCategory = cat;
renderProducts();
}

function renderProducts() {

renderCategories(products);

const query = (document.getElementById('searchInput').value || '').trim().toLowerCase();

const matchCat = selectedCategory === 'All' || p.category === selectedCategory;
const matchQuery = !query ||
p.name.toLowerCase().includes(query) ||
(p.category || '').toLowerCase().includes(query) ||
(p.desc || '').toLowerCase().includes(query);
return matchCat && matchQuery;
});

document.getElementById('productCountText').textContent = `${filtered.length}টি প্রোডাক্ট পাওয়া গেছে`;

</div>`;

const products = getProducts();
const prod = products.find(p => p.id === productId);
if (!prod || !prod.stock) return;

const cart = getCart();
const existing = cart.find(item => item.id === productId);
if (existing) {
existing.qty += 1;
} else {
cart.push({ id: prod.id, name: prod.name, price: Number(prod.price), image: prod.image, qty: 1 });
}
saveCart(cart);
showToast(`"${prod.name}" কার্টে যুক্ত হয়েছে`);
}

function changeQty(productId, delta) {
let cart = getCart();
const item = cart.find(i => i.id === productId);
if (!item) return;
item.qty += delta;
if (item.qty <= 0) {
cart = cart.filter(i => i.id !== productId);
}
saveCart(cart);
}

function updateCartUI() {
const cart = getCart();
const totalQty = cart.reduce((sum, i) => sum + i.qty, 0);
const subtotal = cart.reduce((sum, i) => sum + i.price * i.qty, 0);

document.getElementById('cartBadge').textContent = totalQty;
document.getElementById('cartSubtotal').textContent = `৳${subtotal.toLocaleString()}`;

const container = document.getElementById('cartItems');
const footer = document.getElementById('cartFooter');

if (!cart.length) {
container.innerHTML = `<div class="text-center py-16 text-slate-400 text-sm">আপনার কার্টটি একদম খালি।</div>`;
footer.classList.add('hidden');
return;
}

footer.classList.remove('hidden');
container.innerHTML = cart.map(item => `
<div class="flex items-center gap-3 p-3 rounded-2xl border border-slate-100 bg-white">
<img src="${item.image}" class="w-16 h-16 rounded-xl object-cover bg-slate-100" />
<div class="flex-1 min-w-0">
<h4 class="text-sm font-bold text-slate-900 truncate">${item.name}</h4>
<p class="text-xs font-semibold text-indigo-600 mt-0.5">৳${(item.price * item.qty).toLocaleString()}</p>
<div class="flex items-center gap-2 mt-2">
<button onclick="changeQty('${item.id}', -1)" class="w-6 h-6 rounded-lg bg-slate-100 font-bold text-xs">-</button>
<span class="text-xs font-bold px-1">${item.qty}</span>
<button onclick="changeQty('${item.id}', 1)" class="w-6 h-6 rounded-lg bg-slate-100 font-bold text-xs">+</button>
</div>
</div>
</div>
`).join('');
}

function toggleCart(open) {
document.getElementById('cartDrawer').classList.toggle('hidden', !open);
}

function openCheckoutModal() {
if (!getCart().length) return;
toggleCart(false);
updateCheckoutSummary();
const modal = document.getElementById('checkoutModal');
modal.classList.remove('hidden');
modal.classList.add('flex');
}

function closeCheckoutModal() {
const modal = document.getElementById('checkoutModal');
modal.classList.add('hidden');
modal.classList.remove('flex');
}

function updateCheckoutSummary() {
const subtotal = getCart().reduce((sum, i) => sum + i.price * i.qty, 0);
const delivery = Number(document.getElementById('deliveryArea').value || 120);
document.getElementById('sumSubtotal').textContent = `৳${subtotal.toLocaleString()}`;
document.getElementById('sumDelivery').textContent = `৳${delivery.toLocaleString()}`;
document.getElementById('sumTotal').textContent = `৳${(subtotal + delivery).toLocaleString()}`;
}

function submitOrder(e) {
e.preventDefault();
const cart = getCart();
if (!cart.length) return;

const subtotal = cart.reduce((sum, i) => sum + i.price * i.qty, 0);
const delivery = Number(document.getElementById('deliveryArea').value || 120);
const orderId = 'ORD-' + Math.floor(1000 + Math.random() * 9000);
const payMethod = document.querySelector('input[name="payMethod"]:checked').value;

const newOrder = {
id: orderId,
date: new Date().toLocaleString('bn-BD'),
customer: document.getElementById('custName').value.trim(),
phone: document.getElementById('custPhone').value.trim(),
address: document.getElementById('custAddress').value.trim(),
area: document.getElementById('deliveryArea').selectedOptions[0].text,
payment: payMethod,
items: cart,
subtotal,
delivery,
total: subtotal + delivery,
status: 'Pending'
};

const orders = getOrders();
orders.unshift(newOrder);
saveOrders(orders);
saveCart([]);
closeCheckoutModal();
document.getElementById('checkoutForm').reset();

showToast(`অর্ডার সফল! ID: ${orderId}`);
openTrackModal(orderId);
}

function openTrackModal(prefillId = '') {
const modal = document.getElementById('trackModal');
modal.classList.remove('hidden');
modal.classList.add('flex');
const input = document.getElementById('trackOrderInput');
if (prefillId) {
input.value = prefillId;
trackOrder();
} else {
document.getElementById('trackResult').innerHTML = '';
}
}

function closeTrackModal() {
const modal = document.getElementById('trackModal');
modal.classList.add('hidden');
modal.classList.remove('flex');
}

function trackOrder() {
const id = document.getElementById('trackOrderInput').value.trim().toUpperCase();
const resBox = document.getElementById('trackResult');
if (!id) return;

const order = getOrders().find(o => o.id.toUpperCase() === id);
if (!order) {
resBox.innerHTML = `<div class="p-3.5 rounded-xl bg-rose-50 text-rose-600 font-medium">এই Order ID দিয়ে কোনো অর্ডার পাওয়া যায়নি।</div>`;
return;
}

const badgeColors = {

Shipped: 'bg-indigo-100 text-indigo-800',

resBox.innerHTML = `
<div class="p-4 rounded-2xl bg-slate-50 border border-slate-200 space-y-2">
<div class="flex items-center justify-between">
<span class="font-extrabold text-slate-900">${order.id}</span>

<p class="text-xs text-slate-700"><strong>ঠিকানা:</strong> ${order.address}</p>

<span>মোট বিল (${order.payment})</span>
<span>৳${Number(order.total).toLocaleString()}</span>
</div>
</div>
`;
}
