<script>
const ADMIN_PIN = '1234'; // চাইলে এখান থেকে নিজের পাসওয়ার্ড/পিন পরিবর্তন করুন
const AUTH_KEY = 'auramart_admin_auth';
const PRODUCTS_KEY = 'auramart_products_v1';
const ORDERS_KEY = 'auramart_orders_v1';

function handleLogin(e) {
e.preventDefault();
const pin = document.getElementById('pinInput').value.trim();
if (pin === ADMIN_PIN) {
sessionStorage.setItem(AUTH_KEY, 'true');
showDashboard();
} else {
document.getElementById('loginError').classList.remove('hidden');
}
}

function logout() {
sessionStorage.removeItem(AUTH_KEY);
location.reload();
}

function showDashboard() {
document.getElementById('loginScreen').classList.add('hidden');
document.getElementById('dashboard').classList.remove('hidden');
renderAll();
}

function getProducts() {
return JSON.parse(localStorage.getItem(PRODUCTS_KEY) || '[]');
}

function saveProducts(products) {
localStorage.setItem(PRODUCTS_KEY, JSON.stringify(products));
}

function getOrders() {
return JSON.parse(localStorage.getItem(ORDERS_KEY) || '[]');
}

function saveOrders(orders) {
localStorage.setItem(ORDERS_KEY, JSON.stringify(orders));
}

function switchTab(tab) {
const isOrders = tab === 'orders';
document.getElementById('ordersSection').classList.toggle('hidden', !isOrders);
document.getElementById('productsSection').classList.toggle('hidden', isOrders);

document.getElementById('tabOrders').className = isOrders
? 'px-4 py-2.5 rounded-xl bg-slate-900 text-white text-sm font-bold'
: 'px-4 py-2.5 rounded-xl bg-white text-slate-700 border border-slate-200 text-sm font-bold';

document.getElementById('tabProducts').className = !isOrders
? 'px-4 py-2.5 rounded-xl bg-slate-900 text-white text-sm font-bold'
: 'px-4 py-2.5 rounded-xl bg-white text-slate-700 border border-slate-200 text-sm font-bold';
}

function renderStats() {
const orders = getOrders();
const products = getProducts();
const pending = orders.filter(o => o.status === 'Pending').length;
const revenue = orders
.filter(o => o.status !== 'Cancelled')
.reduce((sum, o) => sum + Number(o.total || 0), 0);

document.getElementById('statTotalOrders').textContent = orders.length;
document.getElementById('statPendingOrders').textContent = pending;
document.getElementById('statTotalRevenue').textContent = `৳${revenue.toLocaleString()}`;
document.getElementById('statTotalProducts').textContent = products.length;
}

function renderOrders() {
const filter = document.getElementById('statusFilter').value;
const orders = getOrders().filter(o => filter === 'All' || o.status === filter);
const container = document.getElementById('ordersList');

if (!orders.length) {
container.innerHTML = `<div class="bg-white p-10 rounded-2xl border border-slate-200/70 text-center text-slate-500 text-sm">
এখনো কোনো অর্ডার পাওয়া যায়নি। কাস্টমার সাইট থেকে অর্ডার করলে এখানে দেখাবে।
</div>`;
return;
}

container.innerHTML = orders.map(o => `
<div class="bg-white p-4 sm:p-5 rounded-2xl border border-slate-200/70 space-y-3">
<div class="flex flex-wrap items-center justify-between gap-2 pb-3 border-b border-slate-100">
<div>
<span class="font-extrabold text-slate-900">${o.id}</span>
<span class="text-xs text-slate-400 ml-2">${o.date}</span>
</div>
<div class="flex items-center gap-2">
<select onchange="updateOrderStatus('${o.id}', this.value)" class="px-3 py-1.5 rounded-xl border border-slate-200 text-xs font-bold bg-slate-50 outline-none">
${['Pending', 'Confirmed', 'Shipped', 'Delivered', 'Cancelled'].map(st =>
`<option value="${st}" ${o.status === st ? 'selected' : ''}>${st}</option>`
).join('')}
</select>
<button onclick="deleteOrder('${o.id}')" class="px-2.5 py-1.5 rounded-xl bg-rose-50 hover:bg-rose-100 text-rose-600 text-xs font-bold">
ডিলিট
</button>
</div>
</div>

<div class="grid grid-cols-1 md:grid-cols-2 gap-3 text-xs sm:text-sm">
<div class="space-y-1">
<p><strong>কাস্টমার:</strong> ${o.customer}</p>
<p><strong>ফোন:</strong> <a href="tel:${o.phone}" class="text-indigo-600 font-bold underline">${o.phone}</a></p>
<p><strong>ঠিকানা:</strong> ${o.address} <span class="text-slate-500">(${o.area})</span></p>
<p><strong>পেমেন্ট:</strong> <span class="font-semibold">${o.payment}</span></p>
</div>

<div class="bg-slate-50 p-3 rounded-xl space-y-1.5">
<p class="text-xs font-bold text-slate-500">অর্ডারকৃত আইটেম:</p>
${(o.items || []).map(item => `
<div class="flex justify-between text-xs">
<span>${item.name} × <strong>${item.qty}</strong></span>
<span class="font-bold">৳${(item.price * item.qty).toLocaleString()}</span>
</div>
`).join('')}
<div class="pt-1.5 border-t border-slate-200 flex justify-between font-extrabold text-slate-900">
<span>সর্বমোট (ডেলিভারিসহ)</span>
<span class="text-indigo-600">৳${Number(o.total).toLocaleString()}</span>
</div>
</div>
</div>
</div>
`).join('');
}

function updateOrderStatus(orderId, newStatus) {
const orders = getOrders().map(o => o.id === orderId ? { ...o, status: newStatus } : o);
saveOrders(orders);
renderStats();
}

function deleteOrder(orderId) {
if (!confirm('এই অর্ডারটি ডিলিট করতে চান?')) return;
const orders = getOrders().filter(o => o.id !== orderId);
saveOrders(orders);
renderAll();
}

function addProduct(e) {
e.preventDefault();
const products = getProducts();
const newProd = {
id: 'PRD-' + Math.floor(1000 + Math.random() * 9000),
name: document.getElementById('pName').value.trim(),
price: Number(document.getElementById('pPrice').value),
oldPrice: document.getElementById('pOldPrice').value ? Number(document.getElementById('pOldPrice').value) : null,
category: document.getElementById('pCategory').value.trim() || 'General',
image: document.getElementById('pImage').value.trim() || 'https://images.unsplash.com/photo-1523275335684-37898b6baf30?auto=format&fit=crop&w=600&q=80',
desc: document.getElementById('pDesc').value.trim(),
stock: true
};

products.unshift(newProd);
saveProducts(products);
e.target.reset();
renderAll();
alert('নতুন প্রোডাক্ট কাস্টমার ওয়েবসাইটে যুক্ত হয়েছে!');
}

function toggleStock(prodId) {
const products = getProducts().map(p => p.id === prodId ? { ...p, stock: !p.stock } : p);
saveProducts(products);
renderProductsAdmin();
}

function deleteProduct(prodId) {
if (!confirm('প্রোডাক্টটি মুছে ফেলতে চান?')) return;
const products = getProducts().filter(p => p.id !== prodId);
saveProducts(products);
renderAll();
}

function renderProductsAdmin() {
const products = getProducts();
const box = document.getElementById('adminProductList');

if (!products.length) {
box.innerHTML = `<p class="text-sm text-slate-500 py-8 text-center">কোনো প্রোডাক্ট নেই। বাম পাশ থেকে যুক্ত করুন।</p>`;
return;
}

box.innerHTML = products.map(p => `
<div class="flex items-center justify-between gap-3 p-3 rounded-2xl border border-slate-100 hover:bg-slate-50/70">
<div class="flex items-center gap-3 min-w-0">
<img src="${p.image}" class="w-14 h-14 rounded-xl object-cover bg-slate-100 shrink-0" />
<div class="min-w-0">
<p class="text-sm font-bold text-slate-900 truncate">${p.name}</p>
<p class="text-xs text-slate-500">${p.category} • <strong class="text-slate-800">৳${Number(p.price).toLocaleString()}</strong></p>
</div>
</div>
<div class="flex items-center gap-2 shrink-0">
<button onclick="toggleStock('${p.id}')" class="px-2.5 py-1.5 rounded-xl text-xs font-bold ${
p.stock ? 'bg-emerald-100 text-emerald-700' : 'bg-amber-100 text-amber-700'
}">
${p.stock ? 'In Stock' : 'Stock Out'}
</button>
<button onclick="deleteProduct('${p.id}')" class="px-2.5 py-1.5 rounded-xl bg-rose-50 hover:bg-rose-100 text-rose-600 text-xs font-bold">
ডিলিট
</button>
</div>
</div>
`).join('');
}

function renderAll() {
renderStats();
renderOrders();
renderProductsAdmin();
}

if (sessionStorage.getItem(AUTH_KEY) === 'true') {
showDashboard();
}
</script>
